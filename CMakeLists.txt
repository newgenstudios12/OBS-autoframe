cmake_minimum_required(VERSION 3.24)
project(obs-autoframe VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AUTOFRAME_BUILD_FILTER "Build OBS filter (requires libobs and OpenCV C++ dev SDK)" ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(autoframe-framing-test tests/framing_test.cpp)
  target_include_directories(autoframe-framing-test PRIVATE src)
  add_test(NAME FramingAndTrackingTests COMMAND autoframe-framing-test)
endif()
if(AUTOFRAME_BUILD_FILTER)
  find_package(libobs CONFIG REQUIRED)
  find_package(OpenCV CONFIG REQUIRED COMPONENTS core imgproc objdetect dnn)
  add_library(obs-autoframe MODULE src/plugin.cpp)
  target_include_directories(obs-autoframe PRIVATE src ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(obs-autoframe PRIVATE OBS::libobs ${OpenCV_LIBS})
  set_target_properties(obs-autoframe PROPERTIES PREFIX "" OUTPUT_NAME "obs-autoframe")
  if(MSVC)
    target_compile_definitions(obs-autoframe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  if(WIN32)
    install(TARGETS obs-autoframe RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
  else()
    install(TARGETS obs-autoframe LIBRARY DESTINATION lib/obs-plugins)
  endif()
endif()
